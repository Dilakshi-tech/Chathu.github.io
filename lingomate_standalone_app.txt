    <script>
        // INITIAL DATA SCHEMA
        const appData = {
            ja: {
                title: "Japanese (Hiragana, Katakana & Kanji)",
                script: [
                    { char: "あ", sub: "a" }, { char: "い", sub: "i" }, { char: "う", sub: "u" },
                    { char: "日", sub: "Sun / Day (Kanji)" }, { char: "本", sub: "Book / Origin (Kanji)" }, { char: "学", sub: "Study (Kanji)" }
                ],
                sentence: { target: ["私", "は", "毎日", "日本語", "を", "勉強します"], hint: "English: I study Japanese every day." }
            },
            fr: {
                title: "French (Alphabet & Diacritics)",
                script: [
                    { char: "A", sub: "ah" }, { char: "É", sub: "e-acute" }, { char: "Ç", sub: "c-cedilla" }
                ],
                sentence: { target: ["J'apprends", "le", "français", "tous", "les", "jours"], hint: "English: I learn French every day." }
            }
        };

        let currentLang = 'ja';

        function switchView(viewId) {
            document.querySelectorAll('.view-container').forEach(v => v.classList.remove('active'));
            document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
            document.getElementById(viewId).classList.add('active');
        }

        function selectLanguage(langKey) {
            currentLang = langKey;
            document.body.className = 'theme-' + langKey;
            renderLanguageData();
        }

        function renderLanguageData() {
            const data = appData[currentLang] || appData['ja'];
            document.getElementById('langTitle').innerText = data.title;
            
            const grid = document.getElementById('scriptGrid');
            grid.innerHTML = '';
            data.script.forEach(s => {
                grid.innerHTML += `<div class="script-card"><div class="script-char">${s.char}</div><div class="script-sub">${s.sub}</div></div>`;
            });

            // Render Scramble
            const avail = document.getElementById('availableZone');
            const assembled = document.getElementById('assembledZone');
            avail.innerHTML = ''; assembled.innerHTML = '';
            
            let shuffled = [...data.sentence.target].sort(() => Math.random() - 0.5);
            shuffled.forEach(w => {
                avail.innerHTML += `<div class="word-tile" onclick="moveTile(this)">${w}</div>`;
            });
            document.getElementById('hintBox').innerText = data.sentence.hint;
        }

        function moveTile(tile) {
            const avail = document.getElementById('availableZone');
            const assembled = document.getElementById('assembledZone');
            if (tile.parentElement.id === 'availableZone') {
                assembled.appendChild(tile);
            } else {
                avail.appendChild(tile);
            }
        }

        function toggleHint() {
            const box = document.getElementById('hintBox');
            box.style.display = box.style.display === 'none' ? 'block' : 'none';
        }

        function selectAvatar(emoji) {
            alert('Selected profile avatar: ' + emoji);
        }

        function updateGoal() {
            const mins = document.getElementById('goalInput').value;
            document.getElementById('hubGoal').innerText = `0 / ${mins} Mins`;
            alert('Daily target updated to ' + mins + ' minutes!');
        }

        // Initialize default view
        renderLanguageData();
    </script>
